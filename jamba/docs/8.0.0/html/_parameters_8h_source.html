547//------------------------------------------------------------------------
548// Parameters::add (called by the builder)
549//------------------------------------------------------------------------
550template<typename T>
551VstParam<T> Parameters::add(VstParamDefBuilder<T> const &iBuilder)
552{
553 auto param = std::make_shared<VstParamDef<T>>(iBuilder.fParamID,
554 iBuilder.fTitle,
555 iBuilder.fUnits,
556 iBuilder.fDefaultValue,
557 iBuilder.fFlags,
558 iBuilder.fUnitID,
559 iBuilder.fShortTitle,
560 iBuilder.fPrecision,
561 iBuilder.fOwner,
562 iBuilder.fTransient,
563 iBuilder.fDeprecatedSince,
564 iBuilder.fConverter);
565
566 if(!iBuilder.fTransient && !iBuilder.fConverter)
567 {
568 DLOG_F(WARNING, "No converter defined for non transient parameter [%d]", iBuilder.fParamID);
569 }
570
571 if(addVstParamDef(param) == kResultOk)
572 return param;
573 else
574 return nullptr;
575}
576
577//------------------------------------------------------------------------
578// Parameters::add (called by the builder)
579//------------------------------------------------------------------------
580template<typename T>
581JmbParam<T> Parameters::add(Parameters::JmbParamDefBuilder<T> const &iBuilder)
582{
583 auto param = std::make_shared<JmbParamDef<T>>(iBuilder.fParamID,
584 iBuilder.fTitle,
585 iBuilder.fOwner,
586 iBuilder.fTransient,
587 iBuilder.fDeprecatedSince,
588 iBuilder.fShared,
589 iBuilder.fDefaultValue,
590 iBuilder.fSerializer);
591
592 if(!iBuilder.fTransient && !iBuilder.fSerializer)
593 {
594 DLOG_F(WARNING, "No serializer defined for parameter [%d] (won't be able to be saved)", iBuilder.fParamID);
595 }
596
597 if(iBuilder.fShared && !iBuilder.fSerializer)
598 {
599 DLOG_F(WARNING, "No serializer defined for parameter [%d] (won't be able to be shared with peer)", iBuilder.fParamID);
600 }
601
602 if(addJmbParamDef(param) == kResultOk)
603 return param;
604 else
605 return nullptr;
606}
