514template<VSTGUI::CViewSubclass TView>
515class ParamAwareView : public ParamAware
516{
517public:
518 // required because we implement other registerCallback methods
519 using ParamAware::registerCallback;
520
521 // defining the change listener callback
522 using ChangeListener = std::function<void(TView *, ParamID)>;
523
524 // Constructor
525 explicit ParamAwareView(TView *iView) : fView(iView), fListener{} {}
526
534 void registerListener(ChangeListener iListener)
535 {
536 fListener = std::move(iListener);
537 }
538
553 template<typename T>
554 GUIVstParam<T> registerCallback(VstParam<T> const &iParamDef,
555 Parameters::ChangeCallback2<TView, GUIVstParam<T>> iChangeCallback,
556 bool iInvokeCallback = false)
557 {
558 auto callback = [view = this->fView, cb2 = std::move(iChangeCallback)] (GUIVstParam<T> &iParam) {
559 cb2(view, iParam);
560 };
561
562 return ParamAware::registerCallback<T>(iParamDef, std::move(callback), iInvokeCallback);
563 }
564
579 template<typename T>
580 GUIJmbParam<T> registerCallback(GUIJmbParam<T> &iParam,
581 Parameters::ChangeCallback2<TView, GUIJmbParam<T>> iChangeCallback,
582 bool iInvokeCallback = false)
583 {
584 auto callback = [view = this->fView, cb2 = std::move(iChangeCallback)] (GUIJmbParam<T> &iParam) {
585 cb2(view, iParam);
586 };
587
588 return ParamAware::registerCallback<T>(iParam, std::move(callback), iInvokeCallback);
589 }
590
594 void onParameterChange(ParamID iParamID) override
595 {
596 if(fListener)
597 fListener(fView, iParamID);
598 }
599
600protected:
601 TView *fView;
602 ChangeListener fListener;
603};
